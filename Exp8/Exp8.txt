
#include <stdio.h>
#include <ctype.h>

int main(void) {
    char key[] = "CIPHER";
    char cipher[27], text[500];
    int used[26] = {0};
    int i, n = 0;

    for (i = 0; key[i]; i++) {
        char c = key[i];
        if (!used[c - 'A']) {
            cipher[n++] = c;
            used[c - 'A'] = 1;
        }
    }

    for (i = 0; i < 26; i++) {
        if (!used[i])
            cipher[n++] = 'A' + i;
    }
    cipher[26] = '\0';

    printf("Plain : ABCDEFGHIJKLMNOPQRSTUVWXYZ\n");
    printf("Cipher: %s\n", cipher);

    printf("Enter plaintext: ");
    fgets(text, sizeof(text), stdin);

    for (i = 0; text[i]; i++) {
        if (isupper((unsigned char)text[i]))
            text[i] = cipher[text[i] - 'A'];
        else if (islower((unsigned char)text[i]))
            text[i] = tolower(cipher[text[i] - 'a']);
    }

    printf("Ciphertext: %s", text);
    return 0;
}
